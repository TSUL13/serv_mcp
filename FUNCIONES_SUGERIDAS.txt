  🔧 EJEMPLO DE IMPLEMENTACIÓN
================================================================================

Para implementar "ver_estadisticas_interfaces", agregarías a server.py:

@mcp.tool()
def ver_estadisticas_interfaces(device_id: str) -> str:
    """
    Consulta estadísticas de todas las interfaces de un dispositivo
    
    Args:
        device_id: System IP del dispositivo
    
    Returns:
        Estadísticas detalladas de interfaces
    """
    try:
        session = get_vmanage_session()
        endpoint = f"/dataservice/device/interface/stats?deviceId={device_id}"
        result = session.get(endpoint)
        
        if 'data' in result:
            interfaces = result['data']
            stats = []
            
            for iface in interfaces:
                stats.append({
                    'interface': iface.get('interface'),
                    'status': iface.get('if-oper-status'),
                    'rx_bytes': iface.get('rx-kbps'),
                    'tx_bytes': iface.get('tx-kbps'),
                    'rx_errors': iface.get('rx-errors'),
                    'tx_errors': iface.get('tx-errors')
                })
            
            return f"Estadísticas de {len(stats)} interfaces:\n{stats}"
        
        return "No se encontraron estadísticas"
        
    except Exception as e:
        return f"Error: {str(e)}"
